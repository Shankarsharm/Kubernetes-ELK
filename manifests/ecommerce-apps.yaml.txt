apiVersion: v1
kind: Namespace
metadata:
  name: ecommerce
---
# Payment Service (Receives HTTP requests and logs them)
apiVersion: apps/v1
kind: Deployment
metadata:
  name: payment-service
  namespace: ecommerce
spec:
  replicas: 1
  selector:
    matchLabels:
      app: payment
  template:
    metadata:
      labels:
        app: payment
    spec:
      containers:
      - name: payment
        image: alpine:latest
        command: ["/bin/sh", "-c"]
        args:
          - |
            apk add --no-cache netcat-openbsd
            while true; do
              echo -e "HTTP/1.1 200 OK\nContent-Type: application/json\n\n{\"status\":\"paid\"}" | nc -l -p 8080
              echo "{\"timestamp\": \"$(date -u +%Y-%m-%dT%H:%M:%SZ)\", \"service\": \"payment-service\", \"action\": \"process_payment\", \"status\": \"200\", \"level\": \"INFO\"}"
            done
        resources:
          limits: { memory: "15Mi", cpu: "20m" }
---
apiVersion: v1
kind: Service
metadata:
  name: payment-service
  namespace: ecommerce
spec:
  selector:
    app: payment
  ports:
    - port: 8080
---
# Order Service (Generates HTTP traffic and logs it)
apiVersion: apps/v1
kind: Deployment
metadata:
  name: order-service
  namespace: ecommerce
spec:
  replicas: 1
  selector:
    matchLabels:
      app: order
  template:
    metadata:
      labels:
        app: order
    spec:
      containers:
      - name: order
        image: alpine:latest
        command: ["/bin/sh", "-c"]
        args:
          - |
            apk add --no-cache curl
            while true; do
              STATUS=$(curl -s -o /dev/null -w "%{http_code}" http://payment-service:8080)
              echo "{\"timestamp\": \"$(date -u +%Y-%m-%dT%H:%M:%SZ)\", \"service\": \"order-service\", \"action\": \"create_order\", \"status\": \"$STATUS\", \"level\": \"INFO\"}"
              sleep 2
            done
        resources:
          limits: { memory: "15Mi", cpu: "20m" }
